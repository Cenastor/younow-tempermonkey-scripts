    const SOUND_DB_NAME = "YouNowChestHelperSoundDB";

    let customOpenSoundName = "";

    function openSoundDatabase() {
        return new Promise((resolve, reject) => {
            try {
                const request = indexedDB.open(SOUND_DB_NAME, SOUND_DB_VERSION);
                request.onupgradeneeded = () => {
                    const db = request.result;
                    if (!db.objectStoreNames.contains(SOUND_STORE_NAME)) {
                        db.createObjectStore(SOUND_STORE_NAME);
                    }
                };
                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            } catch (error) {
                reject(error);
            }
        });
    }

    async function saveCustomOpenSound(file) {
        if (!file) return false;
        try {
            const db = await openSoundDatabase();
            await new Promise((resolve, reject) => {
                const tx = db.transaction(SOUND_STORE_NAME, "readwrite");
                tx.objectStore(SOUND_STORE_NAME).put({
                    blob: file,
                    name: file.name || "Custom sound",
                    type: file.type || "audio/mpeg",
                    updated: Date.now()
                }, SOUND_RECORD_KEY);
                tx.oncomplete = resolve;
                tx.onerror = () => reject(tx.error);
            });
            db.close();
            await loadCustomOpenSound();
            setAction(`Custom sound loaded: ${customOpenSoundName}`);
            return true;
        } catch (error) {
            console.error("[Chest Helper] Sound save failed:", error);
            setAction("Could not save custom sound");
            return false;
        }
    }

    async function loadCustomOpenSound() {
        try {
            const db = await openSoundDatabase();
            const record = await new Promise((resolve, reject) => {
                const tx = db.transaction(SOUND_STORE_NAME, "readonly");
                const request = tx.objectStore(SOUND_STORE_NAME).get(SOUND_RECORD_KEY);
                request.onsuccess = () => resolve(request.result || null);
                request.onerror = () => reject(request.error);
            });
            db.close();
            if (!record || !record.blob) {
                customOpenSoundName = "";
                soundReady = false;
                return false;
            }
            if (customOpenSoundObjectUrl) {
                try { URL.revokeObjectURL(customOpenSoundObjectUrl); } catch (_) {}
            }
            customOpenSoundName = record.name || "Custom sound";
            customOpenSoundObjectUrl = URL.createObjectURL(record.blob);
            soundAudio = new Audio(customOpenSoundObjectUrl);
            soundAudio.preload = "auto";
            soundAudio.volume = clamp(Number(settings.openSoundVolume), 0, 1);
            soundReady = true;
            return true;
        } catch (error) {
            console.warn("[Chest Helper] Could not load saved sound:", error);
            soundReady = false;
            return false;
        }
    }

    async function clearCustomOpenSound() {
        try {
            const db = await openSoundDatabase();
            await new Promise((resolve, reject) => {
                const tx = db.transaction(SOUND_STORE_NAME, "readwrite");
                tx.objectStore(SOUND_STORE_NAME).delete(SOUND_RECORD_KEY);
                tx.oncomplete = resolve;
                tx.onerror = () => reject(tx.error);
            });
            db.close();
        } catch (error) {
            console.warn("[Chest Helper] Could not clear sound:", error);
        }
        if (customOpenSoundObjectUrl) {
            try { URL.revokeObjectURL(customOpenSoundObjectUrl); } catch (_) {}
        }
        customOpenSoundName = "";
        customOpenSoundObjectUrl = null;
        soundAudio = null;
        soundReady = false;
        setAction("Custom chest-open sound removed");
        updateUI();
    }

    async function playCustomOpenSound(force = false) {
        if (!detectOwnBroadcasting()) return false;
        if (!force && !settings.openSoundEnabled) return false;
        if (!soundReady) await loadCustomOpenSound();
        if (!soundReady || !customOpenSoundObjectUrl) {
            setAction("No custom chest-open sound selected");
            return false;
        }
        try {
            if (!soundAudio) soundAudio = new Audio(customOpenSoundObjectUrl);
            soundAudio.volume = clamp(Number(settings.openSoundVolume), 0, 1);
            soundAudio.currentTime = 0;
            const promise = soundAudio.play();
            if (promise && typeof promise.then === "function") await promise;
            setAction(`Playing chest-open sound: ${customOpenSoundName}`);
            return true;
        } catch (error) {
            console.warn("[Chest Helper] Sound playback failed:", error);
            setAction("Sound playback was blocked — use TEST SOUND once");
            return false;
        }
    }

        const file = event?.target?.files?.[0];
        if (!file) return;
        if (!file.type || !file.type.startsWith("audio/")) {
            setAction("Please select an audio file");
            event.target.value = "";
            return;
        }
        await saveCustomOpenSound(file);
        event.target.value = "";
        await playCustomOpenSound(true);
    }

        return number;

            chestLog.slice(0, MAX_LOG_ENTRIES);

    }

    function formatLogDate(timestamp) {
        const date =
            new Date(timestamp);

        if (Number.isNaN(date.getTime())) {
            return "Unknown date";
        }

        return date.toLocaleString(
            undefined,
            {
                dateStyle: "short",
                timeStyle: "medium"
            }
        );
    }

    function escapeHtml(value) {
        return String(value ?? "")
            .replace(/&/g, "&amp;")
            .replace(/</g, "&lt;")
            .replace(/>/g, "&gt;")
            .replace(/"/g, "&quot;")
            .replace(/'/g, "&#039;");
    }

    function renderChestLog() {
        if (!helperPanel) {
            return;
        }

        const container =
            helperPanel.querySelector(
                '[data-field="chest-log"]'
            );

        const count =
            helperPanel.querySelector(
                '[data-field="chest-log-count"]'
            );

        if (!container) {
            return;
        }

        if (count) {
            count.textContent =
                String(chestLog.length);
        }

        if (!chestLog.length) {
            container.innerHTML =
                '<div class="ych-log-empty">No opened chests recorded yet.</div>';
            return;
        }

        container.innerHTML =
            chestLog
                .map((entry, index) => {
                    const likes =
                        Number.isFinite(Number(entry.likes))
                            ? formatLikes(Number(entry.likes))
                            : "—";

                    const threshold =
                        Number.isFinite(Number(entry.threshold))
                            ? formatLikes(Number(entry.threshold))
                            : "—";

                    const delay =
                        Number.isFinite(Number(entry.openingDelay))
                            ? `${Number(entry.openingDelay)}s`
                            : "—";

                    return `
                        <div class="ych-log-entry">
                            <div class="ych-log-top">
                                <strong>Chest #${escapeHtml(entry.cycle)}</strong>
                                <span>${escapeHtml(entry.result || "Opened")}</span>
                            </div>
                            <div class="ych-log-time">${escapeHtml(formatLogDate(entry.timestamp))}</div>
                            <div class="ych-log-details">
                                <span>Likes: <b>${escapeHtml(likes)}</b></span>
                                <span>Target: <b>${escapeHtml(threshold)}</b></span>
                                <span>Delay: <b>${escapeHtml(delay)}</b></span>
                            </div>
                        </div>
                    `;
                })
                .join("");
    }

    function normalizeText(value) {
        return String(
            value || ""
        )
            .replace(

                " "

            .replace(
                /\s+/g,
                " "
            )

    function isVisible(element) {
        if (!element) {
            return false;
        }

        try {
            const style =
                window.getComputedStyle(
                    element
                );

            if (
                style.display ===
                    "none" ||
                style.visibility ===
                    "hidden" ||
                style.opacity ===
                    "0"
            ) {
                return false;
            }

            const rect =
                element.getBoundingClientRect();

            return (
                rect.width > 0 &&
                rect.height > 0
            );
        } catch (_) {
            return false;
        }
    }

    function clamp(value, min, max) {
        return Math.min(max, Math.max(min, Number(value) || 0));
    }

    function formatLikes(value) {
        if (
            value === null ||
            value === undefined ||
            !Number.isFinite(
                Number(value)
            )
        ) {
            return "—";
        }

        return Number(value)
            .toLocaleString(
                "en-US"
            );
    }

    function remainingSeconds(
        end
    ) {
        if (!end) {
            return null;
        }

        return Math.max(
            0,
            Math.ceil(
                (
                    end -
                    Date.now()
                ) / 1000
            )
        );
    }

    function formatTimer(
        seconds
    ) {
        if (
            seconds === null ||
            seconds === undefined
        ) {
            return "—";
        }

        const total =
            Math.max(
                0,
                Math.ceil(
                    seconds
                )
            );

        const minutes =
            Math.floor(
                total / 60
            );

        const secondsPart =
            total % 60;

        if (minutes > 0) {
            return (
                `${minutes}m ` +
                `${String(
                    secondsPart
                ).padStart(
                    2,
                    "0"
                )}s`
            );
        }

        return `${secondsPart}s`;
    }

    function parseYouNowNumber(
        value
    ) {
        if (
            value === null ||
            value === undefined
        ) {
            return null;
        }

        let text =
            String(value)
                .trim()
                .replace(
                    /\u00a0/g,
                    " "
                )
                .replace(
                    /\s+/g,
                    ""
                )
                .toUpperCase();

        if (!text) {
            return null;
        }

        let multiplier = 1;

        if (
            text.endsWith("K")
        ) {
            multiplier = 1000;

            text =
                text.slice(
                    0,
                    -1
                );
        } else if (
            text.endsWith("M")
        ) {
            multiplier = 1000000;

            text =
                text.slice(
                    0,
                    -1
                );
        } else if (
            text.endsWith("B")
        ) {
            multiplier = 1000000000;

            text =
                text.slice(
                    0,
                    -1
                );
        }

        text =
            text.replace(
                /[^0-9.,+-]/g,
                ""
            );

        if (!text) {
            return null;
        }

        if (
            text.includes(".") &&
            text.includes(",")
        ) {
            const dot =
                text.lastIndexOf(".");

            const comma =
                text.lastIndexOf(",");

            if (
                dot > comma
            ) {
                text =
                    text.replace(
                        /,/g,
                        ""
                    );
            } else {
                text =
                    text.replace(
                        /\./g,
                        ""
                    );

                text =
                    text.replace(
                        ",",
                        "."
                    );
            }
        } else if (
            text.includes(",")
        ) {
            const pieces =
                text.split(",");

            if (
                pieces.length > 2 ||
                pieces[
                    pieces.length - 1
                ].length === 3
            ) {
                text =
                    text.replace(
                        /,/g,
                        ""
                    );
            }
        } else if (
            text.includes(".")
        ) {
            const pieces =
                text.split(".");

            if (
                pieces.length > 2 ||
                pieces[
                    pieces.length - 1
                ].length === 3
            ) {
                text =
                    text.replace(
                        /\./g,
                        ""
                    );
            }
        }

        const number =
            parseFloat(
                text
            );

        if (
            !Number.isFinite(
                number
            )
        ) {
            return null;
        }

        return Math.round(
            number *
                multiplier
        );
    }

    function findBroadcastEndButton() {
        const candidates = Array.from(
            document.querySelectorAll(
                'button, [role="button"], input[type="button"], input[type="submit"]'
            )
        ).filter(isVisible);

        for (const element of candidates) {
            const text = normalizeText(element.textContent);
            const aria = normalizeText(element.getAttribute("aria-label"));
            const title = normalizeText(element.getAttribute("title"));
            const value = normalizeText(element.value);

            /* Exact END control only. Do not scan arbitrary page text. */
            if (text === "end" || aria === "end" || title === "end" || value === "end") {
                return element;
            }
        }
        return null;
    }

    function updateBroadcastSafety() {
        const live = detectOwnBroadcasting();

        if (!live) {
            if (ownBroadcastActive || !broadcastDetectionInitialized) {
                ownBroadcastActive = false;
                broadcastDetectionInitialized = true;
                resetAutomationForNonBroadcast();
                if (settings.enabled) {
                    setAction("NOT LIVE — automation stopped (END button not found)");
                }
            }
            return false;
        }

        if (!ownBroadcastActive) {
            ownBroadcastActive = true;

                '[Chest Helper] Chest button click failed:',

            Date.now();

                started <
            5000

                isChestCheckPanelOpen()

                return true;
            }

            await sleep(
                100
            );
        }

        return false;
    }

            let attempt = 0;

            if (

            if (!closeButton) {

                continue;
            }

            const label =
                normalizeText(
                    closeButton.textContent
                ) ||
                closeButton.getAttribute(
                    'aria-label'
                ) ||
                'close';

            const isGotIt =
                label === 'got it';

                    isGotIt

                        : `Chest panel: clicked "${label}" (${attempt + 1}/3)`
                );

                console.warn(
                    '[Chest Helper] Chest panel close click failed:',
                    error
                );

            const started =
                Date.now();

            while (
                Date.now() -
                    started <
                1500

                if (
                    !isChestCheckPanelOpen()
                ) {
                    setAction(
                        `Chest panel CLOSED with "${label}"`
                    );

                    return true;
                }

                await sleep(100);
            }

         * Escape is only a final fallback after the panel's own close

         */
        setAction(
            'Chest panel did not close with its close control — using Escape fallback'

        document.dispatchEvent(
            new KeyboardEvent(
                'keydown',
                {
                    key: 'Escape',
                    code: 'Escape',
                    keyCode: 27,
                    which: 27,
                    bubbles: true,
                    cancelable: true
                }
            )
        );

        await sleep(600);

        const closed =
            !isChestCheckPanelOpen();

        if (closed) {
            setAction(
                'Chest panel CLOSED with Escape'
            );
        } else {
            setAction(
                'Chest panel remains open after Escape'
            );
        }

        return closed;
    }

    /*
     * ============================================================
     * WINNERS PANEL
     * ============================================================
     * YouNow's Winners/Experience overlay is not consistently exposed
     * as one particular class.  Find it from its visible text and then
     * locate the Close control inside that exact element/ancestor.
     */

    function getVisibleElementsWithExactText(text) {
        const wanted = normalizeText(text);

        return Array.from(
            document.querySelectorAll(
                "body *"
            )
        ).filter(element => {
            if (!isVisible(element)) {
                return false;
            }

            return normalizeText(
                element.textContent
            ) === wanted;
        });
    }

    function findClickableCloseIn(root) {
        if (!root) {
            return null;
        }

        const controls = Array.from(
            root.querySelectorAll(
                "button, [role='button'], a, [tabindex], [aria-label], [title]"
            )
        ).filter(isVisible);

        for (const control of controls) {
            const text = normalizeText(
                control.textContent
            );
            const aria = normalizeText(
                control.getAttribute("aria-label")
            );
            const title = normalizeText(
                control.getAttribute("title")
            );

            if (
                text === "close" ||
                aria === "close" ||
                title === "close"
            ) {
                return control;
            }
        }

        /* Some builds render Close as a non-button element. */
        const exactClose =
            getVisibleElementsWithExactText("Close")
                .find(element =>
                    root === element ||
                    root.contains(element)
                );

        if (exactClose) {
            return findClickableAncestor(
                exactClose,
                root
            );
        }

        return null;
    }

    function findClickableAncestor(element, limitRoot = null) {
        let current = element;
        let depth = 0;

        while (
            current &&
            current !== document.body &&
            depth < 8
        ) {
            if (
                current.matches &&
                current.matches(
                    "button, [role='button'], a, [tabindex]"
                ) &&
                isVisible(current)
            ) {
                return current;
            }

            if (
                limitRoot &&
                current === limitRoot
            ) {
                break;
            }

            current = current.parentElement;
            depth++;
        }

        return element && isVisible(element)
            ? element
            : null;
    }

    function findWinnersPanel() {
        /* First choice: the literal Winners! heading. */
        const winnerLabels =
            getVisibleElementsWithExactText("Winners!");

        const candidates = [];

        for (const label of winnerLabels) {
            let current = label.parentElement;
            let depth = 0;

            while (
                current &&
                current !== document.body &&
                depth < 10
            ) {
                const close =
                    findClickableCloseIn(current);

                if (close) {
                    candidates.push({
                        panel: current,
                        close,
                        depth,
                        area: (() => {
                            const r = current.getBoundingClientRect();
                            return r.width * r.height;
                        })()
                    });

                    /* Stop at the first useful containing panel. */
                    break;
                }

                current = current.parentElement;
                depth++;
            }
        }

        /* Fallback for builds that omit the literal Winners! heading. */
        const generic = Array.from(
            document.querySelectorAll(
                '[role="dialog"], .modal, .dialog, [class*="modal"], [class*="drawer"], [class*="panel"]'
            )
        ).filter(isVisible);

        for (const element of generic) {
            const text = normalizeText(
                element.textContent
            );

            if (
                !text.includes("winners") &&
                !text.includes("experience") &&
                !text.includes("reward")
            ) {
                continue;
            }

            const close =
                findClickableCloseIn(element);

            if (close) {
                const r = element.getBoundingClientRect();
                candidates.push({
                    panel: element,
                    close,
                    depth: 99,
                    area: r.width * r.height
                });
            }
        }

        if (!candidates.length) {
            return null;
        }

        candidates.sort((a, b) => {
            if (a.depth !== b.depth) {
                return a.depth - b.depth;
            }

            return a.area - b.area;
        });

        return candidates[0].panel;
    }

    function findWinnerCloseButton(panel) {
        if (!panel) {
            return null;
        }

        const direct =
            findClickableCloseIn(panel);

        if (direct) {
            return direct;
        }

        /* Last-resort search for a visible exact Close element inside panel. */
        const exactClose =
            getVisibleElementsWithExactText("Close")
                .find(element =>
                    panel === element ||
                    panel.contains(element)
                );

        return exactClose
            ? findClickableAncestor(
                exactClose,
                panel
            )
            : null;
    }

    function clickWinnerCloseControl(button) {
        if (!button) {
            return false;
        }

        try {
            if (typeof PointerEvent === "function") {
                button.dispatchEvent(
                    new PointerEvent("pointerdown", {
                        bubbles: true,
                        cancelable: true,
                        button: 0
                    })
                );
            }

            button.dispatchEvent(
                new MouseEvent("mousedown", {
                    bubbles: true,
                    cancelable: true,
                    button: 0
                })
            );

            if (typeof PointerEvent === "function") {
                button.dispatchEvent(
                    new PointerEvent("pointerup", {
                        bubbles: true,
                        cancelable: true,
                        button: 0
                    })
                );
            }

            button.dispatchEvent(
                new MouseEvent("mouseup", {
                    bubbles: true,
                    cancelable: true,
                    button: 0
                })
            );

            button.click();

            return true;
        } catch (error) {
            console.warn(
                "[Chest Helper] Winners close dispatch failed:",
                error
            );

            try {
                button.click();
                return true;
            } catch (_) {
                return false;
            }
        }
    }

    async function closeWinnersPanel() {
        const panel = findWinnersPanel();

        if (!panel) {
            winnerCloseAttempts = 0;
            return true;
        }

        const close =
            findWinnerCloseButton(panel);

        if (!close) {
            winnerCloseAttempts++;
            setAction(
                `Winners panel found — Close control not found (${winnerCloseAttempts})`
            );

            if (winnerCloseAttempts >= 4) {
                setAction(
                    "Winners Close control unavailable — using Escape"
                );

                document.dispatchEvent(
                    new KeyboardEvent("keydown", {
                        key: "Escape",
                        code: "Escape",
                        keyCode: 27,
                        which: 27,
                        bubbles: true,
                        cancelable: true
                    })
                );

                await sleep(500);

                if (!findWinnersPanel()) {
                    winnerCloseAttempts = 0;
                    return true;
                }
            }

            return false;
        }

        winnerCloseAttempts++;

        const clicked =
            clickWinnerCloseControl(close);

        setAction(
            clicked
                ? `Winners Close clicked (${winnerCloseAttempts})`
                : `Winners Close click failed (${winnerCloseAttempts})`
        );

        await sleep(500);

        if (!findWinnersPanel()) {
            winnerCloseAttempts = 0;
            setAction("Winners panel CLOSED");
            return true;
        }

        if (winnerCloseAttempts >= 4) {
            document.dispatchEvent(
                new KeyboardEvent("keydown", {
                    key: "Escape",
                    code: "Escape",
                    keyCode: 27,
                    which: 27,
                    bubbles: true,
                    cancelable: true
                })
            );

            await sleep(500);

            if (!findWinnersPanel()) {
                winnerCloseAttempts = 0;
                setAction("Winners panel CLOSED with Escape");
                return true;
            }
        }

        return false;
    }

    function setNativeValue(
        element,
        value
    ) {
        const prototype =
            Object.getPrototypeOf(
                element
            );

        const descriptor =
            Object.getOwnPropertyDescriptor(
                prototype,
                "value"
            );

        if (
            descriptor &&
            descriptor.set
        ) {
            descriptor.set.call(
                element,
                value
            );
        } else {
            element.value =
                value;
        }

        element.dispatchEvent(
            new Event(
                "input",
                {
                    bubbles: true
                }
            )
        );

        element.dispatchEvent(
            new Event(
                "change",
                {
                    bubbles: true
                }
            )
        );
    }

    function sendChatMessage(
        message
    ) {
        if (!detectOwnBroadcasting()) {
            setAction("Chat blocked — END button not found");
            return false;
        }

        const input =
            getChatInput();

        if (!input) {
            setAction(
                "Chat input not found"
            );

            return false;
        }

        try {
            input.focus();

            setNativeValue(
                input,
                message
            );

            const options = {
                key: "Enter",
                code: "Enter",
                keyCode: 13,
                which: 13,
                bubbles: true,
                cancelable: true
            };

            input.dispatchEvent(
                new KeyboardEvent(
                    "keydown",
                    options
                )
            );

            input.dispatchEvent(
                new KeyboardEvent(
                    "keypress",
                    options
                )
            );

            input.dispatchEvent(
                new KeyboardEvent(
                    "keyup",
                    options
                )
            );

            setAction(
                `Chat sent: ${message}`
            );

            return true;
        } catch (error) {
            console.warn(
                "[Chest Helper] Chat failed:",
                error
            );

            return false;
        }
    }

    /*
     * ============================================================
     * CHAT VARIABLES
     * ============================================================
     *
     * {likes}
     *     = actual likes from the most recent successful chest
     *       panel check.
     *
     * {threshold}
     *     = configured threshold.
     *
     * {lockedLikes}
     *     = likes value locked when threshold was reached.
     *
     * {delay}
     *     = configured opening delay.
     *
     * {cycle}
     *     = current cycle number.
     *
     * {remaining}
     *     = current timer remaining.
     * ============================================================
     */

    function replaceVariables(
        message
    ) {

        /*
         * IMPORTANT:
         *
         * {likes} deliberately uses lastCheckedLikes.
         *
         * It does NOT use threshold.
         * It does NOT use the configured target.
         * It does NOT scan the page.
         */
        const likes =
            lastCheckedLikes !==
                null
                ? lastCheckedLikes
                : currentLikes;

        const remaining =
            remainingSeconds(
                openingTimerEnd
            ) ??
            remainingSeconds(
                chestTimerEnd
            ) ??
            0;

        return String(
            message || ""
        )
            .replace(
                /\{likes\}/gi,
                formatLikes(
                    likes
                )
            )
            .replace(
                /\{threshold\}/gi,
                formatLikes(
                    settings.threshold
                )
            )
            .replace(
                /\{lockedLikes\}/gi,
                formatLikes(
                    lockedLikes
                )
            )
            .replace(
                /\{delay\}/gi,
                String(
                    settings.openingDelaySeconds
                )
            )
            .replace(
                /\{cycle\}/gi,
                String(cycle)
            )
            .replace(
                /\{remaining\}/gi,
                String(remaining)
            );
    }

    function sendOpeningAnnouncement() {
        if (
            !settings.openingChatEnabled ||
            openingChatSent
        ) {
            return;
        }

        const message =
            replaceVariables(
                settings.openingChatMessage
            );

        if (
            sendChatMessage(
                message
            )
        ) {
            openingChatSent =
                true;
        }
    }

    /*
     * ============================================================
     * AUTO INFO
     * ============================================================
     *
     * This timer is completely independent of the chest-opening
     * cycle.
     * ============================================================
     */

    function scheduleAutoInfo() {
        if (
            !settings.autoInfoEnabled
        ) {
            autoInfoNextAt =
                null;

            return;
        }

        autoInfoNextAt =
            Date.now() +
            settings.autoInfoMinutes *
                60 *
                1000;
    }

    function processAutoInfo() {
        if (!detectOwnBroadcasting()) {
            return;
        }

        if (
            !settings.autoInfoEnabled
        ) {
            return;
        }

        if (!autoInfoNextAt) {
            scheduleAutoInfo();

            return;
        }

        if (
            Date.now() <
            autoInfoNextAt
        ) {
            return;
        }

        const message =
            replaceVariables(
                settings.autoInfoMessage
            );

        sendChatMessage(
            message
        );

        scheduleAutoInfo();
    }

    /*
     * ============================================================
     * TIMERS
     * ============================================================
     */

    function startRecheckTimer() {
        if (!settings.timerEnabled) {
            chestTimerEnd = null;
            return;
        }

        chestTimerEnd =
            Date.now() +
            settings.recheckSeconds *
                1000;

        setAction(
            `Recheck timer started: ${settings.recheckSeconds}s`
        );
    }

    function startImmediateRecheck() {
        /* Timer OFF is intentionally a real OFF state, not a 0/1-second timer. */
        if (!settings.timerEnabled) {
            chestTimerEnd = null;
            return;
        }

        chestTimerEnd = Date.now();
        setAction("Immediate chest recheck started");
    }

    function startOpenDelay() {
        openingTimerEnd =
            Date.now() +
            settings.openingDelaySeconds *
                1000;

        setAction(
            `Opening delay started: ${settings.openingDelaySeconds}s`
        );
    }

    /*
     * ============================================================
     * BEGIN FRESH CHEST CYCLE
     * ============================================================
     */

    function beginFreshCycle() {
        /*
         * This is a likes-check/opening cycle, not a completed chest
         * cycle. The completed cycle number advances only after the
         * full opening and cleanup sequence succeeds.
         *
         * Start each chest with a clean last-check value.
         */
        lastCheckedLikes =
            null;

        currentLikes =
            null;

        lockedLikes =
            null;

        likesLockedForCycle =
            false;

        cycleLocked =
            true;

        checkCycleInProgress =
            true;

        likesReadThisCheck =
            false;

        likesCheckDeadline =
            null;

        openingChatSent =
            false;

        openedChestRecordedForCycle =
            false;

        soundPlayedForCycle =
            false;

        preOpenVerification = false;

        chestTimerEnd =
            null;

        openingTimerEnd =
            null;

        winnerCloseAttempts =
            0;

        recoveryStartedAt =
            Date.now();

        setAction(
            `Fresh chest cycle ${cycle} started`
        );

        setState(
            STATES.OPENING_PANEL
        );
    }

    /*
     * ============================================================
     * LOCK LIKES
     * ============================================================
     */

    function lockLikes(
        likes
    ) {
        if (
            likesLockedForCycle
        ) {
            return;
        }

        /*
         * Preserve the actual last checked number.
         */
        lastCheckedLikes =
            likes;

        currentLikes =
            likes;

        lockedLikes =
            likes;

        likesLockedForCycle =
            true;

        cycleLocked =
            true;

        setAction(
            `Locked ${formatLikes(likes)} likes for cycle ${cycle}`
        );
    }

    /*
     * ============================================================
     * STATE MACHINE
     * ============================================================
     */

    async function processState() {

        if (!updateBroadcastSafety()) {
            return;
        }

        if (
            !settings.enabled
        ) {
            state =
                STATES.OFF;

            return;
        }

        /*
         * Independent Auto Info.
         */
        processAutoInfo();

        switch (state) {

            case STATES.WAITING_FOR_CHEST:

                {
                    const chestState = getChestState();

                    /* Preserve existing behavior when the recheck timer is ON. */
                    if (settings.timerEnabled) {
                        if (getChestButton() && !cycleLocked) {
                            beginFreshCycle();
                        }
                        lastObservedChestState = chestState;
                        return;
                    }

                    /* Timer OFF: one check only when the chest becomes AVAILABLE. */
                    if (chestState === "UNAVAILABLE") {
                        offModeCheckedForAvailableChest = false;
                    }

                    const becameAvailable =
                        chestState === "AVAILABLE" &&
                        lastObservedChestState !== "AVAILABLE";

                    if (
                        chestState === "AVAILABLE" &&
                        !offModeCheckedForAvailableChest &&
                        (becameAvailable || lastObservedChestState === "UNKNOWN") &&
                        !cycleLocked
                    ) {
                        offModeCheckedForAvailableChest = true;
                        beginFreshCycle();
                    }

                    lastObservedChestState = chestState;
                }

                return;

            case STATES.OPENING_PANEL:

                /*
                 * This state is entered ONLY when the scheduled check
                 * timer expires. Do not run this continuously.
                 */
                if (!checkCycleInProgress) {
                    return;
                }

                /*
                 * The chest panel is opened/read while the chest is
                 * unavailable too. The panel must remain open until the
                 * real likes value has actually been read.
                 */
                if (
                    isChestCheckPanelOpen()
                ) {
                    setAction(
                        "Chest panel already open — reading likes"
                    );

                    likesCheckDeadline =
                        Date.now() +
                        LIVE_LIKES_CHECK_WINDOW_MS;

                    setState(
                        STATES.READING_LIKES
                    );

                    return;
                }

                if (
                    await openChestPanel(true)
                ) {
                    setAction(
                        "Treasure Chest panel opened"
                    );

                    likesCheckDeadline =
                        Date.now() +
                        LIVE_LIKES_CHECK_WINDOW_MS;

                    setState(
                        STATES.READING_LIKES
                    );
                }

                return;

            case STATES.READING_LIKES:
                {
                    if (
                        likesLockedForCycle
                    ) {
                        return;
                    }

                    const likes =
                        getChestLikes();

                    if (
                        likes === null
                    ) {
                        setAction(
                            "Waiting for live chest likes"
                        );

                        return;
                    }

                    /*
                     * getChestLikes() has already updated:
                     *
                     * lastCheckedLikes
                     * currentLikes
                     *
                     * using the actual chest-panel value.
                     */
                    likesReadThisCheck =
                        true;

                    if (
                        likes <
                        settings.threshold
                    ) {
                        /*
                         * Keep the actual chest panel open briefly while
                         * likes are arriving. YouNow updates .total-likes
                         * live, so the next 500 ms poll can catch the
                         * threshold being reached during this same check.
                         *
                         * This is NOT the recheck timer: once this bounded
                         * window expires, the panel closes and the normal
                         * timer/AVAILABLE-transition rules take over.
                         */
                        if (!likesCheckDeadline) {
                            likesCheckDeadline =
                                Date.now() +
                                LIVE_LIKES_CHECK_WINDOW_MS;
                        }

                        const remaining =
                            Math.max(
                                0,
                                likesCheckDeadline -
                                Date.now()
                            );

                        if (remaining > 0 && isChestAvailable()) {
                            setAction(
                                `Watching live chest likes: ${formatLikes(likes)} / ${formatLikes(settings.threshold)} — ${Math.ceil(remaining / 1000)}s left`
                            );

                            return;
                        }

                        likesCheckDeadline =
                            null;

                        setAction(
                            `Chest likes ${formatLikes(likes)} / ${formatLikes(settings.threshold)} — check complete`
                        );

                        setState(
                            STATES.CLOSING_BELOW_TARGET
                        );

                        return;
                    }

                    /*
                     * The threshold may be reached while the chest is
                     * still unavailable.  Record the real likes value,
                     * close the panel, and wait for the next recheck.
                     * Do not lock the cycle or send an opening message
                     * until the chest itself is AVAILABLE.
                     */
                    if (!isChestAvailable()) {
                        setAction(
                            `Chest likes ${formatLikes(likes)} reached target, but chest is UNAVAILABLE — waiting for next check`
                        );

                        if (!likesReadThisCheck) {
                            return;
                        }

                        const closed =
                            await closeChestPanel();

                        if (!closed) {
                            return;
                        }

                        likesCheckDeadline =
                            null;

                        checkCycleInProgress =
                            false;

                        cycleLocked =
                            false;

                        likesLockedForCycle =
                            false;

                        lockedLikes =
                            null;

                        if (settings.timerEnabled) {
                            startRecheckTimer();
                            setState(STATES.WAITING_FOR_RECHECK);
                        } else {
                            chestTimerEnd = null;
                            preOpenVerification = false;
                            setState(STATES.WAITING_FOR_CHEST);
                        }

                        return;
                    }

                    likesCheckDeadline =
                        null;

                    /* Final verification: do not restart the delay or announcement. */
                    if (preOpenVerification) {
                        lockLikes(likes);
                        const finalClosed = await closeChestPanel();
                        if (!finalClosed) {
                            setAction("Final likes check passed, but chest panel could not close yet — retrying");
                            return;
                        }
                        preOpenVerification = false;
                        checkCycleInProgress = false;
                        setAction(`Final likes check passed: ${formatLikes(likes)} — opening now`);
                        setState(STATES.CLICKING_CHEST);
                        return;
                    }

                    /*
                     * Threshold reached while the chest is AVAILABLE.
                     */
                    lockLikes(
                        likes
                    );

                    /*
                     * Opening announcement uses lastCheckedLikes.
                     */
                    if (
                        settings.openingChatEnabled &&
                        !openingChatSent
                    ) {
                        sendChatMessage(
                            replaceVariables(
                                settings.openingChatMessage
                            )
                        );

                        openingChatSent =
                            true;
                    }

                    /*
                     * Close using the chest panel close control (Got it/Cancel).
                     */
                    const closed =
                        await closeChestPanel();

                    if (!closed) {
                        setAction(
                            "Chest panel could not be closed yet — retrying"
                        );

                        return;
                    }

                    startOpenDelay();

                    setState(
                        STATES.WAITING_FOR_OPEN_DELAY
                    );

                    return;
                }

            case STATES.CLOSING_BELOW_TARGET:
                {
                    if (
                        isChestCheckPanelOpen()
                    ) {
                        const closed =
                            await closeChestPanel();

                        if (!closed) {
                            return;
                        }
                    }

                    checkCycleInProgress =
                        false;

                    likesReadThisCheck =
                        false;

                    likesCheckDeadline =
                        null;

                    cycleLocked =
                        false;

                    preOpenVerification = false;

                    likesLockedForCycle =
                        false;

                    lockedLikes =
                        null;

                    /*
                     * Keep lastCheckedLikes so {likes} continues to
                     * represent the last actual chest check until
                     * the next fresh cycle starts.
                     */

                    if (settings.timerEnabled) {
                        startRecheckTimer();
                        setState(STATES.WAITING_FOR_RECHECK);
                    } else {
                        /* No timer: wait for the next AVAILABLE transition. */
                        chestTimerEnd = null;
                        setState(STATES.WAITING_FOR_CHEST);
                    }

                    return;
                }

            case STATES.WAITING_FOR_RECHECK:

                /*
                 * NEVER close/open the chest panel here. The panel is
                 * handled only by the active timer-driven check cycle.
                 */
                if (
                    checkCycleInProgress
                ) {
                    return;
                }

                if (
                    chestTimerEnd &&
                    Date.now() >=
                        chestTimerEnd
                ) {
                    chestTimerEnd =
                        null;

                    /*
                     * New cycle begins with a clean lastCheckedLikes.
                     */
                    lastCheckedLikes =
                        null;

                    currentLikes =
                        null;

                    lockedLikes =
                        null;

                    likesLockedForCycle =
                        false;

                    cycleLocked =
                        true;

                    checkCycleInProgress =
                        true;

                    likesReadThisCheck =
                        false;

                    setAction(
                        "Recheck timer expired — checking chest again"
                    );

                    setState(
                        STATES.OPENING_PANEL
                    );
                }

                return;

            case STATES.WAITING_FOR_OPEN_DELAY:

                if (
                    isChestModalOpen()
                ) {
                    await closeTreasureChestModal();

                    return;
                }

                if (
                    openingTimerEnd &&
                    Date.now() >=
                        openingTimerEnd
                ) {
                    openingTimerEnd = null;

                    /* Timer OFF: one final live likes check immediately before opening. */
                    if (!settings.timerEnabled && !preOpenVerification) {
                        preOpenVerification = true;
                        checkCycleInProgress = true;
                        likesReadThisCheck = false;
                        likesCheckDeadline = null;
                        /* Unlock the previous value so this is a real fresh read. */
                        likesLockedForCycle = false;
                        lockedLikes = null;
                        lastCheckedLikes = null;
                        currentLikes = null;
                        setAction("Opening delay complete — final chest likes check before opening");
                        setState(STATES.OPENING_PANEL);
                        return;
                    }

                    setAction("Opening delay complete");
                    setState(STATES.CLICKING_CHEST);
                }

                return;

            case STATES.CLICKING_CHEST:

                if (
                    isChestModalOpen()
                ) {
                    await closeTreasureChestModal();

                    return;
                }

                if (
                    !cycleLocked ||
                    !likesLockedForCycle
                ) {
                    setState(
                        STATES.WAITING_FOR_CHEST
                    );

                    return;
                }

                if (
                    settings.openingChatEnabled &&
                    !openingChatSent
                ) {
                    sendOpeningAnnouncement();
                }

                const chestButton =
                    getChestButton();

                if (
                    !chestButton ||
                    !isVisible(
                        chestButton
                    )
                ) {
                    return;
                }

                try {
                    chestButton.click();

                    setAction(
                        `Actual chest clicked — locked likes ${formatLikes(lockedLikes)}`
                    );

                    setState(
                        STATES.CLICKING_OPEN
                    );
                } catch (error) {
                    console.warn(
                        "[Chest Helper] Actual chest click failed:",
                        error
                    );
                }

                return;

            case STATES.CLICKING_OPEN:

                /* Never reread likes here. Play the optional custom
                 * sound immediately before the real Open click. */
                if (
                    !soundPlayedForCycle &&
                    settings.openSoundEnabled
                ) {
                    await playCustomOpenSound();
                    soundPlayedForCycle = true;
                }

                if (
                    clickChestOpenButton()
                ) {
                    setState(
                        STATES.WAITING_FOR_RAIN
                    );

                    return;
                }

                if (
                    Date.now() -
                        stateStartedAt >
                    15000
                ) {
                    setAction(
                        "Open button timeout — continuing recovery"
                    );

                    setState(
                        STATES.WAITING_FOR_RAIN
                    );
                }

                return;

            case STATES.WAITING_FOR_RAIN:

                if (
                    clickMakeItRain()
                ) {
                    setState(
                        STATES.WAITING_FOR_TELL_THEM
                    );

                    return;
                }

                if (
                    Date.now() -
                        stateStartedAt >
                    15000
                ) {
                    setAction(
                        "Make It Rain timeout — continuing"
                    );

                    setState(
                        STATES.WAITING_FOR_TELL_THEM
                    );
                }

                return;

            case STATES.WAITING_FOR_TELL_THEM:

                if (
                    clickTellThem()
                ) {
                    await sleep(
                        500
                    );

                    setState(
                        STATES.WAITING_FOR_CHEST_RESET
                    );

                    return;
                }

                if (
                    Date.now() -
                        stateStartedAt >
                    8000
                ) {
                    setAction(
                        "Tell Them timeout — waiting for chest reset"
                    );

                    setState(
                        STATES.WAITING_FOR_CHEST_RESET
                    );
                }

                return;

            case STATES.WAITING_FOR_CHEST_RESET:

                /*
                 * Do not reread likes.
                 */
                if (
                    getChestState() ===
                    "UNAVAILABLE"
                ) {
                    setAction(
                        "Chest reset detected"
                    );

                    setState(
                        STATES.WAITING_FOR_WINNERS
                    );

                    return;
                }

                if (
                    findWinnersPanel()
                ) {
                    setState(
                        STATES.CLOSING_WINNERS
                    );

                    return;
                }

                if (
                    Date.now() -
                        stateStartedAt >
                    15000
                ) {
                    setAction(
                        "Chest reset timeout — checking Winners"
                    );

                    setState(
                        STATES.WAITING_FOR_WINNERS
                    );
                }

                return;

            case STATES.WAITING_FOR_WINNERS:
                {
                    const winners =
                        findWinnersPanel();

                    if (winners) {
                        winnerCloseAttempts =
                            0;

                        setState(
                            STATES.CLOSING_WINNERS
                        );

                        return;
                    }

                    if (
                        Date.now() -
                            stateStartedAt >
                        12000
                    ) {
                        setAction(
                            "No Winners panel detected — cleanup"
                        );

                        setState(
                            STATES.CLEANUP
                        );
                    }

                    return;
                }

            case STATES.CLOSING_WINNERS:

                if (
                    await closeWinnersPanel()
                ) {
                    setState(
                        STATES.CLEANUP
                    );
                }

                return;

            case STATES.CLEANUP:

                if (
                    isChestModalOpen()
                ) {
                    const closed =
                        await closeTreasureChestModal();

                    if (!closed) {
                        return;
                    }
                }

                if (
                    findWinnersPanel()
                ) {
                    const closed =
                        await closeWinnersPanel();

                    if (!closed) {
                        return;
                    }
                }

                /*
                 * At this point the chest panel and Winners panel are
                 * both confirmed closed, so this is a completed chest.
                 * Record it once, before clearing the cycle values.
                 */
                completeChestCycle();
                addOpenedChestLog();
                renderChestLog();

                openedChestRecordedForCycle =
                    false;

                currentLikes =
                    null;

                lockedLikes =
                    null;

                likesLockedForCycle =
                    false;

                cycleLocked =
                    false;

                preOpenVerification = false;

                openingChatSent =
                    false;

                chestTimerEnd =
                    null;

                openingTimerEnd =
                    null;

                winnerCloseAttempts =
                    0;

                /*
                 * Preserve lastCheckedLikes until the next fresh
                 * cycle. This makes it possible to see the last
                 * actual chest value in the helper/footer.
                 */

                recoveryStartedAt =
                    Date.now();

                /*
                 * Recheck Timer OFF: after a chest has finished opening,
                 * immediately establish the next chest availability check.
                 *
                 * If YouNow has already reset the chest and it is AVAILABLE
                 * right now, seed the OFF-mode edge detector as if the chest
                 * had just been UNAVAILABLE. The next 500 ms poll will then
                 * perform exactly one fresh likes check.
                 *
                 * If it is still UNAVAILABLE, leave the detector at
                 * UNAVAILABLE so the next real AVAILABLE transition will
                 * trigger the next check. No timer is created.
                 */
                const postOpenChestState =
                    getChestState();

                if (!settings.timerEnabled) {
                    chestTimerEnd = null;
                    offModeCheckedForAvailableChest = false;

                    if (postOpenChestState === "AVAILABLE") {
                        lastObservedChestState = "UNAVAILABLE";
                        setAction(
                            `Cycle ${cycle} complete — chest already AVAILABLE; checking again now`
                        );
                    } else {
                        lastObservedChestState =
                            postOpenChestState === "UNAVAILABLE"
                                ? "UNAVAILABLE"
                                : "UNKNOWN";

                        setAction(
                            `Cycle ${cycle} complete — waiting for chest to become AVAILABLE`
                        );
                    }
                } else {
                    setAction(
                        `Cycle ${cycle} complete — waiting for next chest`
                    );
                }

                setState(
                    STATES.WAITING_FOR_CHEST
                );

                return;

            case STATES.OFF:
            default:
                return;
        }
    }

    /*
     * ============================================================
     * RECOVERY
     * ============================================================
     */

    function recovery() {

        if (
            !settings.enabled
        ) {
            return;
        }

        if (
            state ===
                STATES.READING_LIKES ||
            state ===
                STATES.CLOSING_BELOW_TARGET ||
            state ===
                STATES.CLICKING_CHEST ||
            state ===
                STATES.CLICKING_OPEN ||
            state ===
                STATES.WAITING_FOR_RAIN ||
            state ===
                STATES.WAITING_FOR_TELL_THEM
        ) {
            return;
        }

        /*
         * The Open/Cancel Treasure Chest modal is separate from the
         * sidebar check panel. Only Cancel is used here.
         */
        if (
            isChestModalOpen() &&
            (
                state ===
                    STATES.WAITING_FOR_OPEN_DELAY ||
                state ===
                    STATES.CLEANUP
            )
        ) {
            closeTreasureChestModal();

            return;
        }

        if (
            Date.now() -
                recoveryStartedAt >
            120000
        ) {
            currentLikes =
                null;

            lockedLikes =
                null;

            likesLockedForCycle =
                false;

            cycleLocked =
                false;

            openingChatSent =
                false;

            openedChestRecordedForCycle =
                false;

            chestTimerEnd =
                null;

            openingTimerEnd =
                null;

            winnerCloseAttempts =
                0;

            recoveryStartedAt =
                Date.now();

    /*
     * ============================================================
     * PANEL SETTINGS
     * ============================================================
     */

    function applyPanelSettings() {
        if (!helperPanel) {
            return;
        }

        helperPanel.querySelectorAll(
            "[data-setting]"
        ).forEach(
            element => {
                const key =
                    element.dataset.setting;

                if (
                    element.type ===
                    "checkbox"
                ) {
                    element.checked =
                        !!settings[key];
                } else if (
                    key === "openSoundVolume"
                ) {
                    element.value =
                        Math.round(
                            clamp(
                                Number(settings[key]),
                                0,
                                1
                            ) * 100
                        );
                } else {
                    element.value =
                        settings[key] ??
                        "";
                }
            }
        );

        const toggle =
            helperPanel.querySelector(
                '[data-action="toggle"]'
            );

        if (toggle) {
            toggle.textContent =
                settings.enabled
                    ? "ON"
                    : "OFF";

            toggle.classList.toggle(
                "ych-off",
                !settings.enabled
            );
        }
    }

    function savePanelSettings() {
        if (!helperPanel) {
            return;
        }

        helperPanel.querySelectorAll(
            "[data-setting]"
        ).forEach(
            element => {
                const key =
                    element.dataset.setting;

                if (
                    element.type ===
                    "checkbox"
                ) {
                    settings[key] =
                        element.checked;
                } else if (
                    key === "openSoundVolume"
                ) {
                    settings[key] =
                        clamp(
                            Number(element.value) / 100,
                            0,
                            1
                        );
                } else {
                    settings[key] =
                        element.value;
                }
            }
        );

        settings.threshold =
            clampThreshold(
                settings.threshold
            );

        settings.recheckSeconds =
            Math.max(
                1,
                parseInt(
                    settings.recheckSeconds,
                    10
                ) || 1
            );

        settings.openingDelaySeconds =
            Math.max(
                0,
                parseInt(
                    settings.openingDelaySeconds,
                    10
                ) || 0
            );

        settings.autoInfoMinutes =
            Math.max(
                1,
                parseInt(
                    settings.autoInfoMinutes,
                    10
                ) || 1
            );

        saveSettings();

        if (!settings.timerEnabled) {
            chestTimerEnd = null;
            /* Switching the timer OFF starts a new AVAILABLE-edge check, not a timer. */
            lastObservedChestState = "UNKNOWN";
            offModeCheckedForAvailableChest = false;
        }

        scheduleAutoInfo();

        setAction(
            "Settings saved"
        );

        applyPanelSettings();
    }

    function toggleEnabled() {
        if (!settings.enabled && !detectOwnBroadcasting()) {
            setAction("Cannot turn automation ON — END button not found");
            return;
        }

        settings.enabled =
            !settings.enabled;

        saveSettings();

        if (
            !settings.enabled
        ) {
            state =
                STATES.OFF;

            cycleLocked =
                false;

            likesLockedForCycle =
                false;

            chestTimerEnd =
                null;

            openingTimerEnd =
                null;
            autoInfoNextAt = null;
            preOpenVerification = false;
            offModeCheckedForAvailableChest = false;

            setAction(
                "Helper turned OFF"
            );
        } else {
            currentLikes =
                null;

            lockedLikes =
                null;

            likesLockedForCycle =
                false;

            cycleLocked =
                false;

            openingChatSent =
                false;

            openedChestRecordedForCycle =
                false;

            soundPlayedForCycle =
                false;

            state =
                STATES.WAITING_FOR_CHEST;

            recoveryStartedAt =
                Date.now();

            setAction(
                "Helper turned ON"
            );
        }

        applyPanelSettings();
    }

    /*
     * ============================================================
     * PANEL DRAGGING
     * ============================================================
     */

    function clampPosition(
        x,
        y
    ) {
        if (!helperPanel) {
            return {
                x,
                y
            };
        }

        const rect =
            helperPanel.getBoundingClientRect();

        return {
            x: Math.min(
                Math.max(
                    0,
                    x
                ),
                Math.max(
                    0,
                    window.innerWidth -
                        rect.width
                )
            ),

            y: Math.min(
                Math.max(
                    0,
                    y
                ),
                Math.max(
                    0,
                    window.innerHeight -
                        rect.height
                )
            )
        };
    }

    function applyPanelPosition() {
        if (!helperPanel) {
            return;
        }

        if (
            Number.isFinite(
                Number(
                    settings.panelX
                )
            ) &&
            Number.isFinite(
                Number(
                    settings.panelY
                )
            )
        ) {
            const position =
                clampPosition(
                    Number(
                        settings.panelX
                    ),
                    Number(
                        settings.panelY
                    )
                );

            helperPanel.style.left =
                `${position.x}px`;

            helperPanel.style.top =
                `${position.y}px`;

            helperPanel.style.right =
                "auto";
        }
    }

    function beginDrag(
        event
    ) {
        if (
            event.button !== 0 ||
            !helperPanel
        ) {
            return;
        }

        if (
            event.target.closest(
                "button,input,textarea,select"
            )
        ) {
            return;
        }

        const rect =
            helperPanel.getBoundingClientRect();

        dragState.active =
            true;

        dragState.offsetX =
            event.clientX -
            rect.left;

        dragState.offsetY =
            event.clientY -
            rect.top;

        helperPanel.style.left =
            `${rect.left}px`;

        helperPanel.style.top =
            `${rect.top}px`;

        helperPanel.style.right =
            "auto";

        event.preventDefault();
    }

    function dragPanel(
        event
    ) {
        if (
            !dragState.active ||
            !helperPanel
        ) {
            return;
        }

        const position =
            clampPosition(
                event.clientX -
                    dragState.offsetX,

                event.clientY -
                    dragState.offsetY
            );

        helperPanel.style.left =
            `${position.x}px`;

        helperPanel.style.top =
            `${position.y}px`;
    }

    function endDrag() {
        if (
            !dragState.active ||
            !helperPanel
        ) {
            return;
        }

        dragState.active =
            false;

        const rect =
            helperPanel.getBoundingClientRect();

        const position =
            clampPosition(
                rect.left,
                rect.top
            );

        settings.panelX =
            position.x;

        settings.panelY =
            position.y;

        saveSettings();
    }

    function toggleMinimize() {
        settings.minimized =
            !settings.minimized;

        if (
            settings.minimized
        ) {
            settings.maximized =
                false;
        }

        saveSettings();

        applyPanelState();
    }

    function toggleMaximize() {
        settings.maximized =
            !settings.maximized;

        if (
            settings.maximized
        ) {
            settings.minimized =
                false;
        }

        saveSettings();

        applyPanelState();
    }

    function applyPanelState() {
        if (!helperPanel) {
            return;
        }

        helperPanel.classList.toggle(
            "ych-minimized",
            !!settings.minimized
        );

        helperPanel.classList.toggle(
            "ych-maximized",
            !!settings.maximized
        );
    }

    function attachPanelEvents() {
        if (!helperPanel) {
            return;
        }

        helperPanel.querySelector(
            ".ych-header"
        )?.addEventListener(
            "pointerdown",
            beginDrag
        );

        helperPanel.querySelector(
            '[data-action="toggle"]'
        )?.addEventListener(
            "click",
            toggleEnabled
        );

        helperPanel.querySelector(
            '[data-action="save"]'
        )?.addEventListener(
            "click",
            savePanelSettings
        );

        helperPanel.querySelector(
            '[data-action="select-open-sound"]'
        )?.addEventListener(
            "click",
            () => helperPanel.querySelector(
                '[data-action="open-sound-file"]'
            )?.click()
        );

        helperPanel.querySelector(
            '[data-action="open-sound-file"]'
        )?.addEventListener(
            "change",
            handleOpenSoundFile
        );

        helperPanel.querySelector(

    /*
     * ============================================================
     * ELEMENT INFORMATION
     * ============================================================
     */

    function elementInfo(
        element
    ) {
        if (!element) {
            return "Not found";
        }

        const tag =
            element.tagName
                ?.toLowerCase() ||
            "element";

        const id =
            element.id
                ? `#${element.id}`
                : "";

        const classes =
            typeof element.className ===
                "string" &&
            element.className
                ? `.${element.className
                    .trim()
                    .split(/\s+/)
                    .slice(
                        0,
                        2
                    )
                    .join(".")}`
                : "";

        return (
            `${tag}${id}${classes}`
        );
    }

    /*
     * ============================================================
     * UPDATE UI
     * ============================================================
     */

    function updateUI() {
        if (
            !helperPanel ||
            !document.body.contains(
                helperPanel
            )
        ) {
            return;
        }

        const field =
            name =>
                helperPanel.querySelector(
                    `[data-field="${name}"]`
                );

        const setField =
            (
                name,
                value
            ) => {
                const element =
                    field(name);

                if (element) {
                    element.textContent =
                        value;
                }
            };

        setField(
            "status",
            settings.enabled
                ? "ON"
                : "OFF"
        );

        setField(
            "state",
            state
        );

        setField(
            "cycle",
            String(cycle)
        );

        setField(
            "action",
            lastAction
        );

        renderChestLog();

        setField(
            "chest",
            getChestState()
        );

        /*
         * The important distinction:
         *
         * Last Chest Check = actual last chest-panel reading.
         */
        setField(
            "last-check",
            lastCheckedLikes ===
                null
                ? "—"
                : formatLikes(
                    lastCheckedLikes
                )
        );

        setField(
            "likes",
            currentLikes ===
                null
                ? "—"
                : formatLikes(
                    currentLikes
                )
        );

        setField(
            "target",
            formatLikes(
                settings.threshold
            )
        );

        setField(
            "locked",
            lockedLikes ===
                null
                ? "—"
                : formatLikes(
                    lockedLikes
                )
        );

        setField(
            "recheck",
            chestTimerEnd
                ? formatTimer(
                    remainingSeconds(
                        chestTimerEnd
                    )
                )
                : "—"
        );

        setField(
            "opening",
            openingTimerEnd
                ? formatTimer(
                    remainingSeconds(
                        openingTimerEnd
                    )
                )
                : "—"
        );

        setField(
            "auto-info",
            !settings.autoInfoEnabled
                ? "OFF"
                : autoInfoNextAt
                    ? formatTimer(
                        remainingSeconds(
                            autoInfoNextAt
                        )
                    )
                    : "Starting..."
        );

        const soundNameField =
            helperPanel.querySelector(
                '[data-field="open-sound-name"]'
            );

        if (soundNameField) {
            soundNameField.textContent =
                customOpenSoundName ||
                "No sound selected";
            soundNameField.title =
                customOpenSoundName ||
                "No sound selected";
        }

        const soundVolumeField =
            helperPanel.querySelector(
                '[data-field="open-sound-volume"]'
            );

        if (soundVolumeField) {
            soundVolumeField.textContent =
                `${Math.round(
                    clamp(
                        Number(settings.openSoundVolume),
                        0,
                        1
                    ) * 100
                )}%`;
        }

        setField(
            "footer-likes",
            lastCheckedLikes ===
                null
                ? "—"
                : formatLikes(
                    lastCheckedLikes
                )
        );

        const toggle =
            helperPanel.querySelector(
                '[data-action="toggle"]'
            );

        if (toggle) {
            toggle.textContent =
                settings.enabled
                    ? "ON"
                    : "OFF";

            toggle.classList.toggle(
                "ych-off",
                !settings.enabled
            );
        }

        applyPanelState();
    }

    /*
     * ============================================================
     * STARTUP
     * ============================================================
     */

    function start() {

        if (
            !document.body
        ) {
            setTimeout(
                start,
                500
            );

            return;
        }

        installCSS();

        /* Keep the original working v9.48 panel creation method. */
        createHelperPanel();

        /* Audio is optional and initializes only after the panel exists. */
        loadCustomOpenSound()
            .then(() => updateUI())
            .catch(error => {
                console.warn(
                    "[Chest Helper] Custom sound initialization failed:",
                    error
                );
            });

        updateBroadcastSafety();

        if (ownBroadcastActive && settings.enabled) {
            scheduleAutoInfo();
        }

        setAction(
            settings.enabled
                ? (ownBroadcastActive ? "Helper ready — waiting for chest" : "NOT LIVE — automation stopped")
                : "Helper disabled"
        );

        /*
         * Main polling loop.
         *
         * No MutationObserver.
         */
        setInterval(
            async () => {
                /*
                 * processState() contains awaits. Never allow two
                 * state-machine executions to overlap.
                 */
                if (processStateBusy) {
                    return;
                }

                processStateBusy = true;

                try {
                    await processState();

                    recovery();

                    updateUI();
                } catch (error) {
                    console.error(
                        "[Chest Helper] Main loop error:",
                        error
                    );

                    setAction(
                        `Runtime error: ${
                            error?.message ||
                            error
                        }`
                    );
                } finally {
                    processStateBusy = false;
                }
            },
            500
        );

        /*
         * Faster UI refresh for timers.
         */
        setInterval(
            () => {
                try {
                    updateUI();
                } catch (_) {}
            },
            250
        );
    }

    start();

})();
